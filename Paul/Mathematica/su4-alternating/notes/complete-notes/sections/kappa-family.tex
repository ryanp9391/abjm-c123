\section{The pair $\kappa$-family and even bridges}\label{sec:kappa}
%======================================================================

This section solves the problem that the earlier notes called the ``F-site gap'': a $\Lambda$-rotation of a whole A--F pair
(A by $D=1-\kappa\Lambda$, F by $D^{-1}\propto\mathrm{adj}\,D$) by a local transfer-matrix word, with a free continuous parameter
$\kappa$. It exists at the ABJM point of Sec.~\ref{sec:abjm}, for a twist with $z_2=z_3$, on the reference $\bra{W_a^{\otimes L}}$.

\paragraph{Setting.} $h=\tfrac12$; pairwise ABJM inhomogeneities $\theta_{A,n}=\theta_{F,n}+2h$ for every pair $n$ ($\theta_{F,n}$ arbitrary, distinct
or equal); diagonal twist $\Lambda=\mathrm{diag}(z_1,y,y,z_4)$ with $z_2=z_3=:y$; reference $\bra{W_a^{\otimes L}}$ of \eqref{eq:3pt-W}. For pair~1
(sites $1,2$) write $\theta_F=\theta_2$, $\theta_A=\theta_1=\theta_F+2h$ and $u_0:=\theta_F+h=\theta_A-h$. $\hat t_k$ is $t_k$ with exactly
vanishing scalar prefactors $(u-\theta-jh)=0$ dropped. Evidence: exact in \nb{pair-group-rotation.wb} (rerun 2026-10-13: sections 1--6, 8 reproduced, $L=2$ sections 7, 9 partial;
$L=3$ rows of the untwisting sections skipped as too heavy) and in the python of \nb{checks/session-2026-10-12/pairgroup/}, \nb{checks/session-2026-10-12/lpairs/}.

\subsection{Pair-vacuum pattern and the column-support lemma (2603 App.~K style)}\label{sec:kappa-lemma}

\ex\ [\nb{pair-group-rotation.wb} \S1; python \nb{pairvac\_pattern.py}, \nb{scan\_d.py}]. With $\bra S=\sum_m\bra{e_me_m}$ the A--F singlet and
$(\cL_A\cL_F)_{jk}$ the $(j,k)$ auxiliary component of the Lax product on one pair (exact, symbolic in $u,\theta_A,\theta_F$),
\begin{equation}\label{eq:kappa-pattern}
 \bra{e_1e_4}(\cL_A\cL_F)_{jk}(u)=(u-\theta_A)(u-\theta_F)\delta_{jk}\bra{e_1e_4}-h(u-\theta_A)\delta_{k4}\bra{e_1e_j}+h(u-\theta_F)\delta_{j1}\bra{e_ke_4}-h^2\delta_{j1}\delta_{k4}\bra S .
\end{equation}
Seven components vanish identically, $(j,k)\in\{(2,1),(3,1),(4,1),(3,2),(4,2),(4,3),(2,3)\}$, and there are four \emph{bonus} zeros:
$(2,4),(3,4)$ at $u=\theta_A$ and $(1,2),(1,3)$ at $u=\theta_F$. This is the pattern of $\bra Z\cL_{jk}$ in 2603 (six lower-triangular zeros plus bonus zeros), here on the fused pair.

\paragraph{Lemma (column support).} Let $\mathcal D_k$ be the set of $k$-subsets $I\subset\{1,2,3,4\}$ with $1\notin I$, $4\in I$. The rows $I\in\mathcal D_k$ of the $\wedge^k$ auxiliary space
are diagonal on \emph{both} vacua $\bra{e_1}$ (A) and $\bra{e_4}$ (F), with $I$-independent eigenvalues, so pairs $2,\dots,L$ in the vacuum are transparent on $\mathcal D_k$. If the Lax product of
pair~1 at $u_0$ has column support in $\mathcal D_k$, then for \emph{any} twist, hence for any global rotation of $\bra{\vac^{\otimes L}}$ (which intertwines $t^\Lambda$ with $t^{G'}$),
\begin{align}
 \bra{\vac^{\otimes L}}t_k(u_0)&=\lambda_k(u_0)\,\bra{\vac_{\rm pair}}t_k^{(1\text{ pair})}(u_0)\otimes\bra{\vac^{\otimes(L-1)}},\\
 \lambda_k(u)&=\prod_{n\ge2}\Big[\prod_{j=0}^{k-1}(u-\theta_{A,n}-jh)\ (u-\theta_{F,n}-kh)\prod_{j=0}^{k-2}(u-\theta_{F,n}-jh)\Big]
\end{align}
(vanishing factors dropped). The F site at $u_0=\theta_F+(k-1)h$ kills all columns without a $4$; the A site at $u_0=\theta_A-h$ kills all columns containing a $1$. Hence
\begin{itemize}
\item $k=1$: $t_1(\theta_F)$ localises (F-site condition only; any geometry);
\item $k=3$: $\hat t_3(\theta_A-h)$ localises (A-site condition only; any geometry);
\item $k=2$: $t_2(\theta_F+h)$ localises \emph{only if} both conditions hold, i.e.\ \emph{only at $\theta_A-\theta_F=2h$} (scan $d=(\theta_A-\theta_F)/h=-5,\dots,5$ on 4 sites with random twist and random BPS reference, words $t_k(\theta_F+mh)$, $k\le3$, $m=-6,\dots,7$).
\end{itemize}
This is why all searches at generic $\theta$ failed (Sec.~\ref{sec:kappa-generic}) and why the ABJM point, already singled out by $[\HMZ,t_a]=0$ (Sec.~\ref{sec:abjm}), is the place where the pair $\kappa$-family lives.
Every single localising word equals $\lambda_k\times$(the same word on the one-pair chain), exactly [\nb{pairword\_check.py}, Fractions, random rational twist and BPS reference].\footnote{Rerun 2026-10-13 (\nb{pair-group-rotation.wb}, cell~17):
the python check fitted only the constant for \emph{products}. Four of the nine two-factor words tested, $t_1(\theta_F)t_1(\theta_F+h)$, $t_1(\theta_F)\hat t_3(\theta_F+3h)$, $t_1(\theta_F+2h)\hat t_3(\theta_F+h)$ and $\hat t_3(\theta_F+h)\hat t_3(\theta_F+2h)$, do localise and are
proportional to the one-pair word, but with constants $\tfrac{51}{31}$ resp.\ $\tfrac{71}{31}$ (ratios $51/31,\,71/31,\,71/31,\,51/31$) times $\prod\lambda_k$, not $\prod\lambda_k$. $K_\kappa$ below is unaffected: its only product word $X_3X_1$ does have constant $\lambda_1\lambda_3$.}

\paragraph{Operator identities at the ABJM point} \ex\ [\nb{even-bridge-lpairs.wb} \S1a; exact mod $P=268435399$, random product bras, $L=2,3$]:
\begin{equation}
 t_2(u_0)=t_1(u_0)\,t_1(\theta_F),\qquad t_3(u_0)=0\qquad(u_0=\theta_F+h=\theta_A-h),
\end{equation}
the latter after removing the vanishing scalar prefactor ($\hat t_3(u_0)$ is the regular remainder).

\paragraph{The localised algebra on pair~1} \ex\ [\nb{pair-group-rotation.wb}; python \nb{span\_full.py}, \nb{pair\_sym.py}]. All individually localising words (products of length $\le4$ of $t_k(\theta_F+mh)$, $k\le3$, $m=-2,\dots,5$) span, on any reference, exactly
\begin{equation}\label{eq:kappa-alg}
 \{\,1,\ X_1=t_1(\theta_F),\ X_2=t_2(\theta_F+h),\ X_3=\hat t_3(\theta_F+h)\ (=\hat t_3(\theta_A-h)),\ X_3X_1\,\}
\end{equation}
which is $5$-dimensional at the ABJM point and $4$-dimensional at generic $\theta$ (no $X_2$). The same-point generating functions $\sum_kc_kt_k(u_0)$ localise nothing beyond the individually localising terms.
On the pair vacuum with a general twist matrix $G$ (pair-1 chain; $S$ the singlet vector)
\begin{equation}
 \bra{\vac}X_1=2h^2\,e_1\otimes e_4G-h^2G_{41}\,S,\qquad \bra{\vac}X_3=12h^5\,(e_1\,\mathrm{cof}\,G)\otimes e_4,\qquad X_3X_1\propto\rho_{\rm pair}(G^{-T}),
\end{equation}
and $X_2$ is given by the $2\times2$ minors of $G$ with rows $\{2,4\}$, $\{3,4\}$ (no group form). For a diagonal twist the pair-level normalised words
$\xi_1=X_1/(2h^2)$, $\xi_2=-X_2/(2h^4)$, $\xi_3=X_3/(12h^5)$ have eigenvalues on the charged weight states $e_i\otimes e_j$ ($i\ne j$; A index $i$, F index $j$)
\begin{equation}\label{eq:kappa-xi}
 \xi_1\to z_j,\qquad \xi_2\to z_j\,(e_1(z)-z_i-z_j),\qquad \xi_3\to\det\Lambda/z_i,\qquad \xi_3\xi_1\to\det\Lambda\,z_j/z_i ,
\end{equation}
whereas $\rho_{\rm pair}(1-\kappa\Lambda)$ needs $(1-\kappa z_i)/(1-\kappa z_j)$ on these states and $1$ on the zero-weight states. For a twist with generic spectrum the rank-one (product) elements of the five-word span are only
$v(a+bG^{-T})\otimes u$ and $vG^{-T}\otimes u(a+bG)$: there is \emph{no common group rotation of a pair} on any reference with $u\Lambda v^T\neq0$. At $z_2=z_3$ on $W_a$ the span does reach it (next subsection).

\subsection{The exact five-word $K_\kappa$}\label{sec:kappa-word}

\ex\ [\nb{pair-group-rotation.wb}; python \nb{pair\_solve.py}, \nb{word\_verify.py}; pair-level solution symbolic in $z_1,y,z_4,h,a,\kappa$]. With $s=1-\kappa(z_1+z_4)$ and
$\lambda_1=\lambda_1(\theta_F)$, $\lambda_2=\lambda_2(\theta_F+h)$, $\lambda_3=\lambda_3(\theta_F+h)$ (the products over the other pairs of the lemma),
\begin{equation}\label{eq:kappa-K}
 K_\kappa=s^2+\kappa s\Big[\frac{X_1}{h^2\lambda_1}+\frac{X_2}{2h^4y\,\lambda_2}+\frac{X_3}{6h^5y^2\lambda_3}\Big]+\kappa^2\frac{X_3X_1}{24h^7y^2\lambda_1\lambda_3},
\end{equation}
and
\begin{equation}\label{eq:kappa-main}
 \bra{W_a^{\otimes L}}K_\kappa=(1-\kappa z_1)^2\,\bra{W_{a'}\otimes W_a^{\otimes(L-1)}},\qquad a'=a\,\frac{1-\kappa z_4}{1-\kappa z_1}.
\end{equation}
Equivalently $K_\kappa=(1-\kappa z_1)(1-\kappa z_4)\rho_{\rm pair}(1-\kappa\Lambda)$ on $W$-pairs, cf.\ \eqref{eq:3pt-rot}, \eqref{eq:3pt-T}. Exact at $L=2,3$ for $a\in\{\tfrac13,-1\}$ and
$\kappa\in\{\tfrac3{10},-\tfrac75,1/z_4\}$ (the last has $a'=0$). In pair-level normalisation (all $\lambda_k=1$)
\begin{equation}
 K=s^2+\kappa s\big(2\xi_1-\xi_2/y+2\xi_3/y^2\big)+\kappa^2\xi_3\xi_1/y^2 ,
\end{equation}
with coefficients $(s^2,\ \kappa s/h^2,\ \kappa s/(2h^4y),\ \kappa s/(6h^5y^2),\ \kappa^2/(24h^7y^2))$ on $(1,X_1,X_2,X_3,X_3X_1)$; at $\kappa=3/10$, $z=(3/2,5/4,5/4,4/7)$ these are
$(2809/19600,\,159/350,\,636/875,\,1696/4375,\,192/625)$, identical to the Wolfram calibration cell. The word is \emph{unique} (the linear system for the five coefficients has a unique solution), independent of $a$, and local
(the pair-level coefficients depend on $L$ only through $\lambda_k$).

\paragraph{Structure of $K_\kappa$} \ex\ [\nb{even-bridge-epsilon-expansion.wb}, $L=2$; python for $L=3$]. Write, with the eigenvalues $\tau_F=\tau_1(\theta_F)$, $\tau_2(u_0)=x\tau_F$, $\hat\tau_3(u_0)$ of the words,
\begin{equation}\label{eq:kappa-ABC}
 \mathsf A=\frac{\tau_F}{h^2\lambda_1},\qquad \mathsf B=\frac{\tau_2(u_0)}{2h^4y\,\lambda_2},\qquad \mathsf C=\frac{\hat\tau_3(u_0)}{6h^5y^2\lambda_3},\qquad
 K_\kappa(\tau)=s^2+\kappa s(\mathsf A+\mathsf B+\mathsf C)+\tfrac14\kappa^2\,\mathsf A\mathsf C .
\end{equation}
At $\kappa_\star=1/(z_1+z_4)$ ($s=0$) $K_\kappa\propto X_3X_1$, the fixed twist rotation $W_a\to W_{ar}$, $r=z_1/z_4$. At the untwisted point ($\kappa=1$, $y=1$, $s=-1$) one finds $\mathsf A\mathsf C=4$ on all $6$ ($L=2$) and $26$ ($L=3$, python) HW states
($\mathsf A=2e^{i\vartheta}$, $\mathsf C=2e^{-i\vartheta}$). Explanation (exact at $L=2$): $\hat X_3X_1=24h^7\lambda_1\lambda_3\,S=\tfrac9{256}\,S$ with $S$ the one-pair translation (the swap of the two pairs at $L=2$), so $\mathsf A\mathsf C=4$ on translation-invariant states;
this is the shift test $U\propto S$ of Sec.~\ref{sec:abjm}. Therefore
\begin{equation}
 K_1(\tau)=2-\mathsf A-\mathsf B-\mathsf C=2-4\cos\vartheta-\mathsf B .
\end{equation}
$K_1$ vanishes on the vacuum ($\mathsf A=\mathsf C=2$, $\mathsf B=-2$) and on $[L-1,0,L-1]$ ($\mathsf A+\mathsf B+\mathsf C=2$), and is non-zero on other irreps: $L=2$: $10$ and $2$ on the two singlets, $8$ on the zero-weight states of $[0,2,0]$. The characteristic
polynomials of the untwisted $K_1$ operator at $L=2$ in the sectors $(2,0,0,-2)$, $(1,0,0,-1)$, $(0,0,0,0)$ are $-x$, $x^{11}(x-8)$, $x^{24}(x-2)(x-8)^2(x-10)$ \ex.\footnote{The statement ``$K_{\kappa=1}(\tau)=0$ for all non-vacuum states'' made earlier on 2026-10-13 is false:
$K_1$ vanishes exactly on the vacuum tower and $[L-1,0,L-1]$ (i.e.\ $n\le1$, with $n=L-\mu_1$ as in Sec.~\ref{sec:odd-ff}); non-cyclic states can have accidental zeros ($L=2$: $[2,1,0]$, $[0,1,2]$).}

\subsection{Why $z_2=z_3$ is needed}\label{sec:kappa-z2z3}

\ex\ [\nb{pair-group-rotation.wb} \S4; python \nb{negcontrols.py}, \nb{combo\_search.py}, \nb{combo\_L3.py}; exact]. The test on $W_a^{\otimes2}$:
\begin{center}
\begin{tabular}{llcc}
\toprule
point & twist & rank of 5-word span & span + target\\
\midrule
ABJM & $z_2\ne z_3$ & 5 & 6 (fails)\\
generic $\theta$ & $z_2=z_3$ & 4 & 5 (fails)\\
ABJM & $z_2=z_3=y$ & 5 & 5 (\textbf{works})\\
\bottomrule
\end{tabular}
\end{center}
\emph{Mechanism.} At $z_2\ne z_3$ the obstruction lies in the zero-weight directions: $X_1$ has a zero-weight component $\propto E_{22}+E_{33}$ and $X_2$ one $\propto z_3E_{22}+z_2E_{33}$, so the leftover is spanned by these two
(6 directions needed against 5 operators). The cyclic space of $W_a$ on one pair is 6-dimensional (2 weight states plus 4 zero-weight eigenstates) while the algebra is 5-dimensional; at $z_2=z_3$ the zero-weight eigenstate odd under $2\leftrightarrow3$ decouples, the cyclic space becomes 5-dimensional and equals the algebra.
On a generic BPS reference at $z_2=z_3$ the same word gives a pair state of rank~2 (entangled, traceless), $3.5\%$ away from the rotation $d_1d_4\rho(D)$.
Non-localising combinations: at $L=2$ universal combinations of 253 words reach $W_{a'}\otimes W_a$ even at $z_2\ne z_3$ and generic $\theta$ (the two-pair algebra is too rich: meaningless); at $L=3$, ABJM with $z_2\ne z_3$: rank~5, target residual $2.3\cdot10^{-2}$ (fails);
generic $\theta$ with $z_2\ne z_3$: rank~4, fails; ABJM with $z_2=z_3$ reaches; generic $\theta$ with $z_2=z_3$ reaches but the $L^1$-sparsest combination has 80 words with coefficients $10^{-3}\ldots10^{7}$ (no structure; $L$-universality untested) \npy.

\paragraph{Finite twist versus the limit} \npy\ [\nb{task3b.py}, \nb{task3c.py}]. At finite generic twist with $d=z_2-z_3\ne0$ the relative residual of the best word is linear in $d$:
$5.25\cdot10^{-3}$, $5.25\cdot10^{-4}$, $5.25\cdot10^{-5}$, $5.25\cdot10^{-6}$ for $d=10^{-1}$, $10^{-2}$, $10^{-3}$, $10^{-4}$ ($L=3$, $\kappa=\tfrac3{10}$), equal to the least-squares optimum over all five coefficients (the symmetric choice $y\to(z_2+z_3)/2$, $y^2\to z_2z_3$ in the $X_2$ term is the optimum; $y=z_2$ is six times worse): $z_2=z_3$ is a genuine restriction at finite twist.
In the untwisting limit, however, only $z_2-z_3=O(\epsilon)$ is needed: for $z_2-z_3=\tfrac12\epsilon^q$ with $q=1,2,3$ (so $q=1$ has $\varphi_2\ne\varphi_3$) the finite-twist identity is violated by $|{\rm RHS}-{\rm LHS}|\approx0.1\,(z_2-z_3)^2/\epsilon$ (e.g.\ state $\#4$, $[2,0,2]$: $2.6\cdot10^{-4}$, $2.7\cdot10^{-5}$, $2.7\cdot10^{-6}$ for $q=1$), and the \emph{limits are unchanged}
(identical to the $z_2=z_3$ runs). A separation $z_2-z_3\gg\sqrt\epsilon$ would spoil it.
\emph{Caveat} (\nb{even-bridge-sov-formula.wb}): at exactly $z_2=z_3$ the companion frame is singular and the diagonal-twist chain has a $\mathfrak{gl}(2)$ symmetry, so there is no SoV basis; the $K_\kappa$ route has to be combined with SoV at $z_2-z_3=O(\epsilon)$, which suffices for the limit (Sec.~\ref{sec:qform}).

\subsection{The one-pair even bridge and its untwisting limit}\label{sec:kappa-untw}

\ex\ [\nb{pair-group-rotation.wb}, \nb{even-bridge-lpairs.wb}; operator identity; at $L=3$ on all 26 HW states \num{10^{-6}}, limited by cancellation]. Since \eqref{eq:kappa-main} is an operator identity, for any eigenstate $\Psi$ of $t^\Lambda$ with $z_2=z_3$
\begin{equation}\label{eq:kappa-ev}
 \braket{W_{a'}\otimes W_a^{\otimes(L-1)}}{\Psi}=\frac{K_\kappa(\tau_\Psi)}{(1-\kappa z_1)^2}\,\braket{W_a^{\otimes L}}{\Psi},\qquad K_\kappa(\tau)=\text{\eqref{eq:kappa-ABC}}.
\end{equation}
At $\kappa=1/z_4$ (vacuum pair, $a'=0$), with $\Sigma=\mathsf A+\mathsf B+\mathsf C$ and $\Pi=\tfrac14\mathsf A\mathsf C$ the eigenvalues of $2\xi_1-\xi_2/y+2\xi_3/y^2$ and $\xi_3\xi_1/y^2$,
\begin{equation}
 \braket{\vac\otimes W_a^{\otimes(L-1)}}{\Psi}=\frac{z_1^2-z_1\Sigma+\Pi}{(z_1-z_4)^2}\,\braket{W_a^{\otimes L}}{\Psi}.
\end{equation}

\paragraph{Untwisting recipe} \num{O(\epsilon)}\ [\nb{even-bridge-lpairs.wb} \S2, rerun 2026-10-13; python \nb{untwist.py}, \nb{task2.py}]. $z=1+\epsilon\varphi$ with $\varphi_2=\varphi_3$ (standard $\varphi=(0.9,0.35,0.35,-1.1)$),
$\theta_{F,n}=\epsilon\theta^0_n$ ($\theta^0=(0,0.37,-0.52)$), $\theta_A=\theta_F+2h$; $\Psi_\epsilon$ is the twisted eigenvector in the weight sector connected to the untwisted HW state $\Psi_0$ (all non-degenerate, overlap $\to1$).
The right-hand side of \eqref{eq:kappa-ev} is a $0\cdot\infty$ product: $K_\kappa=O(1)$ or smaller, $\braket{W_a^{\otimes L}}{\Psi_\epsilon}=O(\epsilon^n)$ and $(1-\kappa z_1)^2=O(\epsilon^2)$ (details below).
\begin{enumerate}
\item[(i)] $\kappa=1/z_4$, $a=-1$: $\to\braket{\vac\otimes W_{-1}^{\otimes(L-1)}}{\Psi_0}$ ($M=L-1$). $L=2$: singlets $-1/\sqrt{40}$, $1/\sqrt{24}$ (errors $1.6,2.0\cdot10^{-3}$ at $\epsilon=10^{-2}$, decreasing linearly), $[1,0,1]$: $0$, vacuum: $1$.
\item[(ii)] $\kappa\to1$ with $a'-a\to-1$: $\kappa=1/(z_1-a(z_1-z_4))$ with $a=\tfrac13$ (then $a'-a=-1$ exactly), or the linear prescription $\kappa=1-c\epsilon$, $a=(\varphi_1-c)/(\varphi_1-\varphi_4)$, for which
$a'-a=a\kappa(z_1-z_4)/(1-\kappa z_1)\to a(\varphi_1-\varphi_4)/(c-\varphi_1)=-1$. The finite HW shift $e^{-aJ}$ gives
$\braket{W_{a'}\otimes W_a^{\otimes(L-1)}}{\Psi_0}=\braket{W_{-1}\otimes\vac^{\otimes(L-1)}}{\Psi_0}=\braket{\vac^{\otimes(L-1)}\otimes W_{-1}}{\Psi_0}$ (cyclicity), i.e.\ $M=1$.
\end{enumerate}
In (ii) the shift parameter $a$ stays \emph{finite}, which is exactly what the highest-weight trick of Sec.~\ref{sec:3pt-hw} lacked ($b\to\infty$). The limits are independent of direction and of $c$: for $\varphi=(.9,.35,.35,-1.1)$, $(-.7,1.3,1.3,.4)$ and $(1.5,-.6,-.6,.2)$ with
$c=0.3$ and $c=2.0$, states $0,4,16,17,18,20,22,25$, all six runs converge as $O(\epsilon)$ to the same values (e.g.\ state~16, target $-0.171126$: errors $3.4,19,3.4,11,1.8,32\cdot10^{-5}$ at $\epsilon=10^{-4}$). $\kappa=1/z_4$ is the member $c=\varphi_4$ ($a=1$, $a'=0$).

\paragraph{Orders in $\epsilon$ (which states contribute)} \num{10^{-6}}\ [\nb{even-bridge-epsilon-expansion.wb}; python \nb{scan\_eps.py}]. With $n=L-\mu_1$ for weight $(\mu_1,0,0,-\mu_1)$ one has $w:=\braket{W_a^{\otimes L}}{\Psi_\epsilon}=O(\epsilon^n)$ and $w\equiv0$ for every other weight,

\subsection{The admissible set at $z_2=z_3$}\label{sec:admiss-set}

\paragraph{Charged-sector structure} \ex. The words commute with the Cartan torus; on the 12 charged states $e_a\otimes e_b$ ($a\ne b$) their eigenvalues are \eqref{eq:kappa-xi}. \subsection{The A-frozen twisted-vacuum limit}\label{sec:admiss-frozen}

\ex\ [\nb{reference-admissibility-Kkappa.wb}; chain $L=2,3$ mod $p$, python \nb{r6\_l3.py}]. At $z_2=z_3$ the diagonal-frame image of $\bra{0_w}$ degenerates (it exists only for $z_2\ne z_3$; at $z_2=z_3$ it is the rescaled limit $v=(c/\delta)(0,-1,1,0)+O(1)$, whose $O(1)$ part is lost): $v\to(0,1,-1,0)$, a \emph{frozen} A site on the $\mathfrak{gl}(2)$-odd direction, projectively independent of $w$, with $u=u_w$ generic.
On this limit reference a common pair rotation exists, but it is an F-site rotation only: the A site is an eigenvector of $\Lambda$, so ``rotation'' means $u\to u(1-\kappa\Lambda)^{-1}$, $v\to v\cdot\mathrm{const}$. Choosing $c_4=c_5=0$ above, with $e_1'=z_1+y+z_4$, $e_2'=yz_1+yz_4+z_1z_4$,
\begin{equation}\label{eq:admiss-KF}
 K^F_\kappa=\big[1-\kappa e_1'+\kappa^2e_2'\big]+2\kappa X_1+8\kappa^2X_2\qquad(h=\tfrac12;\ \text{general }h:\ \tfrac{\kappa}{2h^2}X_1+\tfrac{\kappa^2}{2h^4}X_2)
\end{equation}
at pair level (chain level: $c_1+(2\kappa/\lambda_1)X_1+(8\kappa^2/\lambda_2)X_2$; $X_3$, $X_3X_1$ are not needed). It is the one-site analogue of 2603's word $1-\kappa t_++\kappa^2t_0$. On an F index $b$ it acts as
$\prod_{j\in\{1,y,4\}}(1-\kappa z_j)/(1-\kappa z_b)=1-\kappa(e_1'-z_b)+\kappa^2(e_2'-z_b(e_1'-z_b))$; its eigenvalue on a state is $P^F_\kappa(\tau)=1-\kappa(e_1'-\xi_1)+\kappa^2(e_2'-\xi_2)$ with $\xi_1=\tau_{X_1}/2h^2$, $\xi_2=-\tau_{X_2}/2h^4$,
a Q-form via $x$, $\tau_F$ of Sec.~\ref{sec:kappa-qform} ($c_3=c_4=0$ in $P_K=c_0+c_1\tau_F+c_2x\tau_F+c_3y+c_4y\tau_F$ of \nb{even-bridge-sov-formula.wb}). Verified on the chain at $L=2,3$ (mod $p$), on the twisted-vacuum limit, the companion-vacuum limit and a generic $(u_1,u_2,u_2,u_4)$ for $\kappa\in\{\tfrac3{10},-\tfrac75,\tfrac74,\tfrac29\}$.
The twisted vacuum is also admissible at its loci $w_1=z_1$, $w_1=z_4$, $w_2=w_3$, $w_4=z_4$ and combinations (rank of the span $1,2$ or $3$); every ``yes'' among the twisted-vacuum rows has a frozen A site.

\subsection{Cross pairings: why $W_a$ is the only useful reference}\label{sec:admiss-cross}

\ex\ [\nb{reference-admissibility-Kkappa.wb}; two-block Borel invariants of Sec.~\ref{sec:polforms}]. The cross-block pairings that make a two-block polarisation sensitive to the $s>0$ states are, for the blocks $(v,u)$ and $(vD,uD^{-1})$ with $D=\mathrm{diag}(c-\varphi_j)$ and $\pi_C=v|_C\cdot u|_C$,
\begin{equation}
 p_{23}=\sum_C\frac{\pi_C}{c-\varphi_C},\qquad p_{32}=\sum_C\pi_C\,(c-\varphi_C).
\end{equation}
Every admissible reference except the $W$-type has all $\pi_C=0$, hence $p_{23}\equiv p_{32}\equiv0$; $W_a$ is admissible exactly because $\pi_1=-\pi_4\ne0$ is allowed at $z_2=z_3$. So ``cross pairing $\ne0$'' and ``$K$-admissible'' are compatible only for the $W$-type family.
For the A-frozen polarisation $a_2=a_3=0$, $m_v=0$, $p_{23}=p_{32}=0$ and only $d_2,d_3,m_u$ survive; with the structure of Sec.~\ref{sec:polforms} ($F$ polynomial in $a_2,a_3,d_2,d_3,m_v,m_u,p_{23},p_{32}$ with $e_8+e_9=s$) it can only see the $e_8=e_9=0$ monomials: \emph{it is not the generic two-block polarisation} the 2603-type construction was meant to produce
(inference from the polarisation forms, not recomputed). For the non-frozen admissible family $(C)$: $a_2=1$, $d_2=u_4$, $m_v=p_2(\varphi_1-\varphi_2)$, $m_u\ne0$, but again $p_{23}=p_{32}=0$. If an A-frozen polarisation is acceptable, the usable word is $K^F_\kappa$ of \eqref{eq:admiss-KF}.
Open: the Q-form of $\braket{0_w}{\Psi}$ is not connected to any admissible $K$; a $z_2\ne z_3$ generalisation of $K_\kappa$ needs at least $11$ additional localised operators (none known).

\subsection{Untwisting of a rotated pair: $D=\mathrm{diag}(c-\varphi_j)$}\label{sec:admiss-untw}

\ex\ [python \nb{untw.py}; exact]. With $z=1+\epsilon\varphi$, $w=1+\epsilon\omega$, $\kappa=1-c\epsilon$ one has $1-\kappa z_j=\epsilon(c-\varphi_j)+O(\epsilon^2)$, so the rotation in the limit is $D=\mathrm{diag}(c-\varphi_j)$, \emph{not} $c+\varphi_j$
(the latter corresponds to $\kappa=1+c\epsilon$ or $\varphi\to-\varphi$). For $\varphi=(\tfrac9{10},\tfrac7{20},\tfrac7{10},-\tfrac{11}{10})$ ($\varphi_2\ne\varphi_3$), $\omega=(\tfrac7{10},-\tfrac3{10},\tfrac15,\tfrac12)$, $c=\tfrac45$ the deviation of the limit of $T_\kappa$ from
$(\hat vD)\otimes(\hat uD^{-1})$, with $\hat v_j=\prod_{m\ne j}(\omega_1-\varphi_m)/(\varphi_j-\varphi_m)$ and $\hat u_j=\prod_{m\le3}(\varphi_j-\omega_m)$, is
\begin{center}
\begin{tabular}{lcccc}
\toprule

\midrule

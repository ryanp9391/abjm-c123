\section{Chain, conventions and transfer matrices}
%======================================================================

\paragraph{Chain.} $2L$ sites (``$L$ pairs''), alternating A F A F $\dots$, periodic. A sites carry the $\mathbf 4$
($\rho_A(E_{ij}) = e_{ij}$, group acts by $g$), F sites the $\bar{\mathbf 4}$ ($\rho_F(E_{ij}) = -e_{ji}$, group acts by
$g^{-T}$). In ABJM language: $\mathrm{tr}(Y^{A_1}Y^\dagger_{B_1}\cdots Y^{A_L}Y^\dagger_{B_L})$. Length $L=2,3,4$ means
$4,6,8$ sites and state spaces of dimension $256, 4096, 65536$. $P_{ab}=\sum e_{ij}\otimes e_{ji}$ (same-type sites),
$K_{ab}=\sum e_{ij}\otimes e_{ij}$ (A--F pair; $4\times$ the singlet projector). $h=\tfrac12$ throughout.

\paragraph{Fundamental transfer matrix.} $\cL_s(u)=\sum_{ij}e_{ij}\otimes\big((u-\theta_s)\delta_{ij}+h\,\rho_s(E_{ij})\big)$,
i.e.\ $\cL_A = u-\theta+hK$, $\cL_F=u-\theta-hP$; $T=\cL_1\cdots\cL_{2L}$, twist on the right,
$t_1(u)=\mathrm{tr}\,T(u)G$. $t_a$ are the quantum-minor fusions (rows antisymmetrised, column $k$ at $u-(k-1)h$).

\paragraph{One-pass $t_2$ and $t_3$} [\nb{dual-lax-t3.wb}, exact]. From the one-site quantum minors:
\begin{align}
 t_3(u) &= \prod_s g^{(3)}_s(u)\;\mathrm{tr}\big[\Lb_1(u)\cdots\Lb_{2L}(u)\,\det G\,G^{-T}\big],\\
 \Lb_A &= (u-\theta+h)-hP,\qquad g^{(3)}_A=(u-\theta-h)(u-\theta-2h),\nonumber\\
 \Lb_F &= (u-\theta-3h)+hK,\qquad g^{(3)}_F=(u-\theta)(u-\theta-h),\nonumber
\end{align}
in the sign gauge $(-1)^k$ on the dual index; and
\begin{equation}
 t_2(u)=\prod_s g^{(2)}_s(u)\;\mathrm{tr}\big[\cL^{(2)}_1\cdots\cL^{(2)}_{2L}\,\wedge^2 G\big],\quad
 \cL^{(2)}_s=(u-\tilde\theta_s)+h\sum_{ab}D(e_{ab})\otimes\rho_s(E_{ab}),
\end{equation}
$D$ = derivation action on $\wedge^2\mathbb C^4$, $\tilde\theta_A=\theta$, $\tilde\theta_F=\theta+h$, $g^{(2)}_A=u-\theta-h$,
$g^{(2)}_F=u-\theta$. Both checked identically against the quantum minors at $2L=2,4$ for companion, diagonal,
generic and trivial twist. Relative A/F shift: $0$ for $t_1$, $h$ for $t_2$, $4h$ for $t_3$ (the earlier ``naive'' dual
Lax had the right $P/K$ structure but no relative shift). $\tb_a$ denotes the reduced transfer matrices (without the
scalar $\prod g_s$).

\section{The ABJM point}
%======================================================================

[\nb{abjm-chain.wb}, \nb{abjm-bridges.wb}] Minahan--Zarembo Hamiltonian (arXiv:0806.3951; Bak--Rey arXiv:0807.2063),
prefactor $\lambda^2/2$ dropped:
\begin{equation}
 \HMZ=\sum_{l=1}^{2L}\big(2-2P_{l,l+2}+P_{l,l+2}K_{l,l+1}+K_{l,l+1}P_{l,l+2}\big).
\end{equation}
Hermitian, $\mathfrak{su}(4)$-invariant, invariant under the pair shift $S$. Spectrum at $L=2$: $E=0\ (\times84,\ [2,0,2]$
BPS), $4,\ 8\ (\times120),\ 12\ (\times30),\ 16\ (\times20),\ 20$.

\paragraph{Result.} For pair-periodic inhomogeneities ($\theta_A$ on A sites, $\theta_F$ on F sites),
$[\HMZ,\tb_a(u)]=0$ for $a=1,2,3$ \emph{iff} $\theta_A-\theta_F=2h$ (exact at $L=2$ with $u$ and $d=\theta_A-\theta_F$
symbolic: the gcd of all commutator coefficients is $\propto d-2h$; numerically $10^{-15}$ at $L=3$). Moreover
\begin{equation}
 U(u)=\tb_1(\theta_F+u)\,\tb_3(\theta_A-h+u),\qquad U(0)=\tfrac1{16}S,\qquad \HMZ=U(0)^{-1}U'(0)+12\quad(L=2,\ \text{exact}).
\end{equation}
In general $U(0)=c\,S$ with $c=h^{4}[x(-x-4h)]^2$ at $L=2$, $x=\theta_F-\theta_A$; it vanishes at $\theta_A-\theta_F\in\{0,4h\}$.

\paragraph{Remark (superseded results).} The earlier bridge computations used $\theta_A=\theta_F$, where $U(0)=0$, the
shift drops out of the $t$-algebra, $\HMZ$ does not commute with the $t$'s, and an odd-momentum HW partner is
degenerate in $t_1,t_2,t_3$. All results below are at the ABJM point $\theta_A=2h$, $\theta_F=0$, untwisted and
homogeneous unless stated.

\section{Highest-weight eigenstates and Q-functions}
%======================================================================

\paragraph{States} [\nb{abjm-LR-symmetry.wb}]. All HW, cyclic ($S=1$) joint eigenstates of $\tb_1,\tb_3$: $6,\,26,\,152$
at $L=2,3,4$; all $\HMZ$ eigenstates ($\le3.5\cdot10^{-13}$), joint spectrum non-degenerate (gap $\ge0.0065$). Saved in
\nb{data/abjm-hw-states.wxf}; eigenvalue polynomials $\tb_{1,2,3}(u)$ in \nb{data/abjm-tau.wxf}.

\paragraph{Q-functions} [\nb{abjm-Qfunctions.wb}]. On the vacuum $T_{ii}|0\rangle=V_i|0\rangle$ with
$V_1=(u-\theta_A+h)^L(u-\theta_F)^L$, $V_2=V_3=(u-\theta_A)^L(u-\theta_F)^L$, $V_4=(u-\theta_A)^L(u-\theta_F-h)^L$, and
$\tau_a^{\rm vac}(u)=\sum_{i_1<\dots<i_a}V_{i_1}(u)V_{i_2}(u-h)\cdots$, so
$\sum_a(-1)^a\tau_aD^a=(1-V_1D)\cdots(1-V_4D)$ on the vacuum, $D=e^{-h\partial_u}$. Baxter equation in polynomial gauge:
\begin{equation}
 \sum_{a=0}^{4}(-1)^a\,\tau_a(u)\,Q(u+h-ah)\prod_{b=a}^{3}V_4(u-bh)=0 .
\end{equation}
$Q_1,Q_2,Q_3$ = its polynomial solutions ordered by degree; $Q_{12}=\mathrm{Cas}(Q_1,Q_2)/\mathrm{Cas}_{\rm vac}$,
$Q_{123}=\mathrm{Cas}(Q_1,Q_2,Q_3)/\mathrm{Cas}_{\rm vac}$. For all 184 states
$\deg(Q_1,Q_{12},Q_{123})=(K_1,K_2,K_3)$ with $K_2=L-\tfrac{p+r}2-q$, $K_1=\tfrac{L-r+K_2}2$, $K_3=\tfrac{L-p+K_2}2$
for Dynkin $[p,q,r]$. Saved in \nb{data/abjm-Qfunctions.wxf}.

\section{Discrete symmetries: LR, parity and their product}
%======================================================================

[\nb{abjm-LR-symmetry.wb}, \nb{abjm-other-symmetries.wb}] Two operators commute with $\HMZ$:
$\cP=$ (Weyl reversal $e_a\mapsto e_{5-a}$)$\,\circ\,$(one-site shift), and $\cQ=$ (signed Weyl $g$:
$e_1\to e_4,\ e_2\to e_3,\ e_3\to e_2,\ e_4\to-e_1$)$\,\circ\,$(reflection $s\mapsto 2L+1-s$). Numerically
($10^{-16}$):
\begin{equation}
 \cP\,\tb_1(u)\,\cP^{-1}=\tb_3(u+h),\qquad \cQ\,\tb_1(u)\,\cQ^{-1}=\tb_1(2h-u),\qquad \cP^2=S,\quad \cQ S\cQ^{-1}=S^{-1}.
\end{equation}
$\cP$ is the ABJM version of 2603's LR map ($\mathbf 4\leftrightarrow\bar{\mathbf 4}$). Symmetric states, with the
Q-function form verified exactly (fixed shifts) on all 184 states:

\begin{center}
\begin{tabular}{lll}
\toprule
symmetry & eigenvalues & Q-functions\\
\midrule
LR ($\cP$) & $\tau_1(u)=\tau_3(u+h)$ & $Q_1(u)\propto Q_{123}(u+h)$\\
P ($\cQ$) & $\tau_1(u)=\tau_1(2h-u)$ & $Q_1(u)\propto Q_{123}(2h-u)$\\
LP ($\cP\cQ$) & $\tau_1(u)=\tau_3(3h-u)$ & $Q_1(u)\propto Q_1(h-u)$\\
\bottomrule
\end{tabular}
\end{center}
$\tau_2(u)=\tau_2(c-u)$ holds exactly for the P and LP states. Every non-LR state has an LR partner
($Q_1^{\rm partner}=Q_{123}(u+h)$). Counts:

\begin{center}
\begin{tabular}{cccccccccc}
\toprule
$L$ & states & LR & P & LP & all three & LR only & P only & LP only & none\\
\midrule
2 & 6 & 4 & 6 & 4 & 4 & 0 & 2 & 0 & 0\\
3 & 26 & 8 & 18 & 12 & 8 & 0 & 10 & 4 & 4\\
4 & 152 & 24 & 60 & 40 & 20 & 4 & 40 & 20 & 68\\
\bottomrule
\end{tabular}
\end{center}
$\cQ$-eigenstates have eigenvalue $(-1)^p$ (in this sign convention).

\section{Structure constants of the note's configuration}
%======================================================================

[\nb{abjm-bridges.wb}, \nb{abjm-LR-symmetry.wb}] Bra for bridge $\ell$: first $\ell$ sites O$_2$ fields (A: $e_1$, F: $e_4$),
the rest O$_3$ fields (A: $e_1-e_4$, F: $e_4+e_1$); $C_\ell=\langle B_\ell|\Psi\rangle/\|\Psi\|$.

\paragraph{$L=2$ at the ABJM point.}
\begin{center}
\begin{tabular}{lll}
\toprule
state & $E_{\rm MZ}$ & nonzero $C_\ell$\\
\midrule
$[2,0,2]$ vacuum & 0 & all $1$\\
$[0,2,0]$ & 16 & ---\\
$[1,0,1]$ parity pair & 12, 12 & $C_1=C_3=(1\mp i\sqrt3)/\sqrt{48}$, $|C_1|=1/\sqrt{12}$\\
singlet & 20 & $C_2=-1/\sqrt{40}$\\
singlet & 4 & $C_2=1/\sqrt{24}$\\
\bottomrule
\end{tabular}
\end{center}
The odd-bridge formula $\langle B_1|\Psi\rangle=\frac{1-z_4/(c_1\tau_1(\theta_1))}{1-z_4/z_1}\langle W^L_{-1}|\Psi\rangle$
holds through the untwisting limit at the ABJM point ($10^{-11}$--$10^{-14}$, generic inhomogeneous deformation, no shift
operator needed).

\paragraph{When does $C_\ell$ vanish ($L=2,3,4$).}
(i) R-charge: unless $\Psi\in[p,0,p]$, and unless $L-p\le\ell\le L+p$. (ii) Bridge parity (kinematic, whole HW cyclic
sector): if $L+p$ is odd, $C_\ell=0$ unless $\ell\equiv L+p\pmod 2$. (iii) No LR selection rule: non-LR states have
nonzero $C_\ell$ (e.g.\ the $L=2$ adjoint pair). (iv) Isolated zeros at $L=4$: the LR $[1,0,1]$ state at $E=32$
(all $\ell$) and the LR $[2,0,2]$ state at $E=8$ (only even $\ell$ survive).

\section{Other three-point configurations}
%======================================================================

[\nb{abjm-3pt-configs.wb}] Two-block BPS bras: block 1 on $\ell$ sites with $(a_1,f_1)$, block 2 on the rest with
$(a_2,f_2)$, $a_i\cdot f_i=0$. Up to $GL(4)$ and rescalings only six classes; with block 1 $=(e_1,e_4)$, block 2:
generic $(e_1-e_4,e_4+e_1)$ (= the note's), $a_2\cdot f_1=0$: $(e_2,e_1+e_3)$, $a_1\cdot f_2=0$: $(e_4+e_2,e_3)$,
both zero $(e_2,e_3)$, $a_2=a_1$: $(e_1,e_3)$, $f_2=f_1$: $(e_2,e_4)$. Bridge starting on A or F. Whole multiplets probed
by random global rotations.

\paragraph{LR.} No configuration has an LR rule at $L=3,4$ (only an isolated $L=2$ case). ``Only non-LR states couple''
occurs (8, 14, 24 configurations at $L=2,3,4$) but not uniformly in $L$.

\paragraph{P: length-independent selection rules.} With the \emph{same polarisations at every $L$}, the following give
$C=0$ for every non-P state, verified at operator level on the cyclic sector ($\langle B|X_P(u)|r\rangle=0$ for cyclic $r$,
$10^{-16}$ at $L=3,4$; empty at $L=2$ where every cyclic state is P-symmetric):
\begin{center}
\begin{tabular}{lll}
\toprule
block 2 & bridge $\ell$ & start\\
\midrule
generic (note's $W_{-1}$) & $1$ or $2L-1$ & A or F\\
$a_2\cdot f_1=0$ & $1$ & A\\
$a_1\cdot f_2=0$ & $1$ & F\\
both cross pairings zero, $(e_2,e_3)$ & $2$ or $2L-2$ & A or F\\
\bottomrule
\end{tabular}
\end{center}
Control: the note's configuration at $\ell=2$ fails ($10^{-2.4}$). No uniform LP rule.

\section{Factorised bras annihilated by the symmetry differences}
%======================================================================

[\nb{abjm-LR-factorized.wb}, \nb{abjm-other-symmetries.wb}] $X_{\rm LR}(u)=\tb_1(u)-\tb_3(u+h)$,
$X_{\rm P}(u)=\tb_1(u)-\tb_1(2h-u)$, $X_{\rm LP}(u)=\tb_1(u)-\tb_3(3h-u)$.
\begin{itemize}
\item LR, $L=2$ untwisted: $\ker X_{\rm LR}$ (left) has dim $136/256$ ($X_{\rm LR}$ is linear in $u$); product states in it:
only the BPS orbit. Twisted with $\Lambda=\mathrm{diag}(z_1,z_2,1/z_2,1/z_1)$ ($z_1z_4=z_2z_3=1$, so the $t_3$ twist
$\det\Lambda\,\Lambda^{-1}$ is the Weyl image of $\Lambda$): 36 LR-symmetric eigenstates, kernel dim 36, product states only
$e_a\otimes e_{5-a}$ (vacuum and Weyl images); the note's $W_a$ only for $a=0$.
\item P: \emph{every} two-site state $(v\otimes w)^{\otimes L}$ is annihilated by $X_{\rm P}$ ($10^{-16}$ at $L=2,3,4$),
the analogue of integrable two-site boundary states.
\item LP: product solutions at $L=2$ (arbitrary product states in a rotated SU(2)$\times$SU(2) sector) fail at $L=3,4$.
\end{itemize}

\section{Companion-twist SoV basis at four sites and BPS expansions}
%======================================================================

[\nb{sov-companion-L4.wb}] Four sites A F A F, companion twist $G_c(z)$, generic $\theta$. The 256 words
\begin{equation}
 \langle a_1,k_2,a_3,k_4|=\langle0|\,t_{a_1}(\theta_1-h)\,t_1(\theta_2)^{k_2}\,t_{a_3}(\theta_3-h)\,t_1(\theta_4)^{k_4},\qquad a_i,k_i\in\{0,1,2,3\},
\end{equation}
form a basis (exact rank 256), with factorised overlaps $\langle\Lambda|\Psi\rangle=\tau_{a_1}(\theta_1-h)\tau_1(\theta_2)^{k_2}
\tau_{a_3}(\theta_3-h)\tau_1(\theta_4)^{k_4}\langle0|\Psi\rangle$. Expansions $\langle B|=\sum_\Lambda c_\Lambda\langle\Lambda|$ of BPS
states $v\otimes w\otimes v\otimes w$, $v\cdot w=0$ (number of nonzero $c_\Lambda$):
\begin{center}
\begin{tabular}{ll}
\toprule
BPS state & $\#\{c_\Lambda\neq0\}$\\
\midrule
vacuum & 1\\
F sites only, $v=e_1$, $w\in e_1^\perp$ general & 9 \ ($a_1=a_3=0$, $k_2,k_4\le2$)\\
F only, $\exp(\kappa E_{34})$ & 4 \ ($k\le1$; not a product, F--F exchange term)\\
F only, $\exp(\kappa E_{24})$ & 5\\
A only, $\exp(\kappa E_{12})$ & 13 \ ($a\le2$, $k\le1$)\\
